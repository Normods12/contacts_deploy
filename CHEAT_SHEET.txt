📦 PREREQUISITES (One-time setup - 10 min)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
□ Install Docker Desktop: https://www.docker.com/products/docker-desktop/
□ Create Docker Hub account: https://hub.docker.com/signup
□ Get your Render DATABASE_URL from PostgreSQL dashboard


🚀 QUICK DEPLOYMENT (7 minutes)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

STEP 1: Login to Docker Hub (30 seconds)
───────────────────────────────────────────────────────────────
docker login
# Enter your Docker Hub username and password


STEP 2: Build Image (3 minutes)
───────────────────────────────────────────────────────────────
cd path/to/contacts-final

docker build -t YOUR_DOCKERHUB_USERNAME/contacts-backend:latest .

# Example: docker build -t sujal123/contacts-backend:latest .


STEP 3: Push to Docker Hub (2 minutes)
───────────────────────────────────────────────────────────────
docker push YOUR_DOCKERHUB_USERNAME/contacts-backend:latest

# Example: docker push sujal123/contacts-backend:latest


STEP 4: Deploy on Render (2 minutes)
───────────────────────────────────────────────────────────────
1. Go to: https://dashboard.render.com/
2. New + → Web Service
3. "Deploy an existing image from a registry"
4. Image URL: YOUR_DOCKERHUB_USERNAME/contacts-backend:latest
5. Add environment variables:
   
   DATABASE_URL = postgres://user:pass@host:port/database
   JWT_SECRET = your_secret_key_12345
   FRONTEND_URL = http://localhost:3000
   PORT = 8080

6. Click "Create Web Service"
7. Wait 1 minute → DONE! ✅


🔄 UPDATE YOUR APP (2 minutes)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
docker build -t YOUR_USERNAME/contacts-backend:latest .
docker push YOUR_USERNAME/contacts-backend:latest

Then in Render: Manual Deploy → Clear build cache & deploy

→ Check DATABASE_URL starts with postgres://

💡 PRO TIPS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
• First build takes 3-5 min (downloads Java/Maven)
• Subsequent builds are faster (~1 min - cached)
• Always use :latest tag for auto-updates
• Check Render logs if deployment fails
• DATABASE_URL must start with postgres:// (NOT jdbc:)
